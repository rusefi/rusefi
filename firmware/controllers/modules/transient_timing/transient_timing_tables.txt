! [tag:transient_timing] Signed ignition correction during TPS events, see TransientTiming::update().
#define TRANSIENT_TIMING_SIZE 4
custom transient_timing_mode_e 1 bits, U08, @OFFSET@, [0:1], "Off", "Log only", "Apply"
transient_timing_mode_e transientTimingMode;Off, Log only (computed, not applied) or Apply.
uint8_t[TRANSIENT_TIMING_SIZE] transientTimingTpsChangeBins;TPS change reported by the AE detector (deltaTps).;"%", 1, 0, 0, 100, 0

int8_t[TRANSIENT_TIMING_SIZE] autoscale transientTimingClosingValues;Degrees during a closing event, then hold and decay. Positive advances.;"deg", 0.5, 0, -30, 30, 1
uint16_t transientTimingHoldMs;Time at full value after the event.;"ms", 1, 0, 0, 5000, 0
uint16_t transientTimingDecayMs;Time back to zero after the hold.;"ms", 1, 0, 0, 5000, 0
uint8_t transientTimingMaxAdvance;Largest advance added. None while knock control retards.;"deg", 1, 0, 0, 20, 0
uint8_t transientTimingMaxRetard;Largest retard removed.;"deg", 1, 0, 0, 30, 0
