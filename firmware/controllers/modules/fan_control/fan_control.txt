	bit cranking
	bit notRunning
	bit disabledWhileEngineStopped;Engine stopped
	bit brokenClt;Broken CLT
	bit enabledForAc;Enable for AC
	bit enabledForAcPressure;Enable for AC pressure
	bit hot;Above hot threshold
	bit cold;Below cold threshold
	bit disabledBySpeed
	bit m_state;On
	bit pwmActive;PWM mode active
	uint8_t radiatorFanStatus
	float fanSpeedTarget;Fan speed target\n0% = fan off, 100% = full fan. Pulled from the PWM curve plus any A/C boost or inhibit override, before soft-start ramping.
	float fanSpeedApplied;Fan speed applied\n0% = fan off, 100% = full fan. Same as Fan speed target but after soft-start ramping - commanded demand, not a measured fan speed.
	float pwmAppliedPwm;PWM duty applied\nThe raw PWM duty cycle sent to the pin, after off/full-speed duty mapping. May be inverted vs fan speed (min/max can be set either way to match hardware).
